<script>
function getData(key){
return JSON.parse(localStorage.getItem(key)||'[]');
}
function setData(key,data){
localStorage.setItem(key,JSON.stringify(data));
}
function show(id){
document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
document.getElementById(id).classList.add('active');
}
function addItem(type,inputId){
let input=document.getElementById(inputId);
let value=input.value.trim();
if(!value){alert('اكتب الاسم أولاً');return;}
let data=getData(type);
data.push(value);
setData(type,data);
input.value='';
render();
}
function renderList(type,id){
let data=getData(type);
document.getElementById(id).innerHTML=data.map((v,i)=>
'<p>'+ (i+1)+'. '+v+
' <button onclick="removeItem(\''+type+'\','+i+')">حذف</button></p>'
).join('')||'لا توجد بيانات حتى الآن';
}
function removeItem(type,index){
let data=getData(type);
data.splice(index,1);
setData(type,data);
render();
}
function saveAttendance(){
let name=document.getElementById('attendanceName').value.trim();
let status=document.getElementById('attendanceStatus').value;
if(!name){alert('اكتب اسم الطالب');return;}
let data=getData('attendance');
data.push({name,status,date:new Date().toLocaleDateString('ar')});
setData('attendance',data);
document.getElementById('attendanceName').value='';
render();
}
function saveMemorization(){
let name=document.getElementById('memorizationName').value.trim();
let part=document.getElementById('memorizationPart').value.trim();
let review=document.getElementById('reviewPart').value.trim();
if(!name){alert('اكتب اسم الطالب');return;}
let data=getData('memorization');
data.push({name,part,review});
setData('memorization',data);
document.getElementById('memorizationName').value='';
document.getElementById('memorizationPart').value='';
document.getElementById('reviewPart').value='';
render();
}
function render(){
renderList('students','studentsList');
renderList('rings','ringsList');
renderList('teachers','teachersList');
document.getElementById('studentCount').textContent=getData('students').length;
document.getElementById('ringCount').textContent=getData('rings').length;
document.getElementById('teacherCount').textContent=getData('teachers').length;
document.getElementById('attendanceList').innerHTML=getData('attendance').map(x=>'<p>'+x.name+' — '+x.status+' — '+x.date+'</p>').join('');
document.getElementById('memorizationList').innerHTML=getData('memorization').map(x=>'<p>'+x.name+' — الحفظ: '+x.part+' — المراجعة: '+x.review+'</p>').join('');
}
render();
</script>
